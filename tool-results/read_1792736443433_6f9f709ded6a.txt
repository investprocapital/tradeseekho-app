    21→interface Signal {
    22→  id: string
    23→  symbol: string
    24→  signalType: string
    25→  entry: string
    26→  entry2: string | null
    27→  stopLoss: string

    30→  tp3: string | null
    31→  note: string | null
    32→  screenshot: string

    34→  // different field names. All point to the same base64 data URL.
    35→  chart_image_url?: string
    36→  chart_snapshot_url?: string
    37→  status: string // active | tp1_hit | tp2_hit | tp3_hit | sl_hit
    38→  tp1HitAt: string | null

    40→  tp3HitAt: string | null

    44→  hideLevels?: boolean

    47→
    48→async function fetchSignals() {
    49→  const res = await fetch("/api/signals")
    50→  if (!res.ok) throw new Error("failed")
    51→  return res.json() as Promise<{ signals: Signal[] }>
    52→}
    53→
    54→// Resolve the chart image URL. Checks all three field names the API returns
    55→// (screenshot / chart_image_url / chart_snapshot_url) so any client — web or
    56→// Flutter — gets the captured chart image. Returns "" when none exist.
    57→function chartImage(s: Signal): string {
    58→  return s.screenshot || s.chart_image_url || s.chart_snapshot_url || ""
    59→}

   474→/**
   475→ * Fallback "chart" shown in the detail modal when a signal has no captured
   476→ * screenshot image (chart_image_url was null). Renders a styled signal-card
   477→ * with the price-level ladder so the client never shows an empty dark box.
   478→ */
   479→function FallbackSignalCard({ s }: { s: Signal }) {
   480→  const isBuy = s.signalType === "BUY"
   481→  const accent = isBuy ? "#00D09C" : "#FF6B6B"
   482→  const nums = [
   483→    { label: "SL", val: parseFloat(s.stopLoss), color: "#FF6B6B", hit: !!s.slHitAt },
   484→    { label: "ENTRY", val: parseFloat(s.entry), color: "#FFFFFF", hit: false },
   485→    { label: "TP1", val: s.tp1 ? parseFloat(s.tp1) : NaN, color: "#00D09C", hit: !!s.tp1HitAt },
   486→    { label: "TP2", val: s.tp2 ? parseFloat(s.tp2) : NaN, color: "#22D3EE", hit: !!s.tp2HitAt },
   487→    { label: "TP3", val: s.tp3 ? parseFloat(s.tp3) : NaN, color: "#3B82F6", hit: !!s.tp3HitAt },
   488→  ].filter((n) => Number.isFinite(n.val))
   489→
   490→  const min = Math.min(...nums.map((n) => n.val))
   491→  const max = Math.max(...nums.map((n) => n.val))
   492→  const range = Math.max(max - min, 1e-6)
   493→  const pad = range * 0.15
   494→  const lo = min - pad
   495→  const hi = max + pad
   496→  const pct = (v: number) => `${((hi - v) / (hi - lo)) * 100}%`
   497→
   498→  return (
   499→    <div
   500→      className="relative w-full overflow-hidden"
   501→      style={{ background: "linear-gradient(180deg,#0B1B2E 0%,#0A1929 100%)" }}
   502→    >
   503→      <div style={{ height: 4, background: accent }} />
   504→      <div className="p-4">
   505→        {/* Header: pair + BUY/SELL pill */}
   506→        <div className="flex items-center justify-between">
   507→          <span className="text-2xl font-extrabold text-white">{pairLabel(s.symbol)}</span>
   508→          <span
   509→            className="rounded-full px-3 py-1 text-xs font-bold"
   510→            style={{ background: accent, color: "#0A1929" }}
   511→          >
   512→            {s.signalType}
   513→          </span>
   514→        </div>
   515→        {/* Ladder */}
   516→        <div className="relative mt-4 mb-2" style={{ height: 200 }}>
   517→          <div className="absolute left-3 top-0 bottom-0 w-0.5 bg-[#1E3A5F]" />
   518→          {nums.map((n) => (
   519→            <div
   520→              key={n.label}
   521→              className="absolute flex items-center"
   522→              style={{ top: pct(n.val), left: 0, right: 0 }}
   523→            >
   524→              <div className="h-0.5 flex-1" style={{ background: n.color }} />
   525→              <div
   526→                className="ms-auto me-3 rounded px-2 py-0.5 text-[11px] font-bold"
   527→                style={{ background: n.color, color: "#0A1929" }}
   528→              >
   529→                {n.label}: {n.val} {n.hit ? "✅" : ""}
   530→              </div>
   531→            </div>
   532→          ))}
   533→        </div>
   534→        <div className="mt-1 text-center text-[10px] text-white/40">
   535→          TradeSeekho PK · LEARN TRADE GROW
   536→        </div>
   537→      </div>
   538→    </div>
   539→  )
   540→}
