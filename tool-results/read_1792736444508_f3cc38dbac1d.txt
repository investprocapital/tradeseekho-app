   310→// Price Alert card — shown on home dashboard below Live Signals.
   311→// Shows a summary of active alerts + opens the Price Alert sheet on click.
   312→function PriceAlertCard() {
   313→  const setPriceAlertsOpen = useStore((s) => s.setPriceAlertsOpen)
   314→  // Fetch ad settings to check if Price Alerts is enabled
   315→  const { data: adSettings } = useQuery<{ priceAlertsEnabled?: boolean }>({
   316→    queryKey: ["ad-settings"],
   317→    queryFn: async () => {
   318→      const res = await fetch("/api/ads/config", { cache: "no-store" })
   319→      if (!res.ok) return { priceAlertsEnabled: true }
   320→      return res.json()
   321→    },
   322→    staleTime: 60_000,
   323→  })
   324→  // If admin has disabled Price Alerts, hide the card entirely
   325→  if (adSettings?.priceAlertsEnabled === false) return null
   326→  return (
   327→    <motion.button
   328→      initial={{ opacity: 0, y: 8 }}
   329→      animate={{ opacity: 1, y: 0 }}
   330→      onClick={() => setPriceAlertsOpen(true)}
   331→      className="mt-2 flex w-full items-center gap-3 rounded-xl border-2 border-gold/30 bg-gradient-to-r from-gold/5 to-transparent p-3 text-start transition hover:border-gold/60 hover:from-gold/10 active:scale-[0.99]"
   332→    >
   333→      <span className="inline-flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-gold text-white">
   334→        <Bell className="h-5 w-5" />
   335→      </span>
   336→      <div className="min-w-0 flex-1">

   362→function LevelBox({
   363→  category, index, lessons, onOpen, compact = false,
   364→}: {
   365→  category: CategoryDTO; index: number; lessons: { id: string; passed: boolean; orderInCategory: number; title: { en: string; ur: string; hi: string; ar: string } }[]; onOpen: () => void; compact?: boolean
   366→}) {
   367→  const lang = useStore((s) => s.lang)
   368→  const Icon = LEVEL_ICONS[category.icon || ""] || BookOpen
   369→  const total = lessons.length
   370→  const done = lessons.filter((l) => l.passed).length
   371→  const pct = total ? Math.round((done / total) * 100) : 0
   372→
   373→  if (compact) {
   374→    return (
   375→      <motion.button
   376→        initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3, delay: index * 0.06 }}
   377→        onClick={onOpen}
   378→        className="group relative flex flex-col items-center gap-2 overflow-hidden rounded-2xl border-2 p-3 text-center transition-all hover:-translate-y-0.5"
   379→        style={{ borderColor: `${category.color || "#00c853"}80`, background: `linear-gradient(135deg, ${category.color || "#00c853"}25, ${category.color || "#00c853"}08)` }}
   380→      >
   381→        {/* Colored top accent bar */}
   382→        <div className="absolute left-0 right-0 top-0 h-1" style={{ background: category.color || "#00c853" }} />
   383→        <span className="inline-flex h-10 w-10 shrink-0 items-center justify-center rounded-xl text-white shadow" style={{ background: category.color || "#00c853" }}>
   384→          <Icon className="h-5 w-5" />
   385→        </span>
   386→        <div className="w-full">
   387→          <h3 className={`text-sm font-black leading-tight ${lang === "ur" || lang === "ar" ? "font-urdu" : ""}`}>{category.name[lang] || category.name.en}</h3>
   388→          <span className="mt-0.5 inline-block rounded-full bg-background/70 px-1.5 py-0.5 text-[9px] font-bold text-muted-foreground">{done}/{total}</span>
   389→        </div>
   390→        <div className="h-1 w-full overflow-hidden rounded-full bg-muted">
   391→          <div className="h-full rounded-full transition-all" style={{ width: `${pct}%`, background: category.color || "var(--brand)" }} />
   392→        </div>
   393→        <span className="inline-flex items-center gap-1 rounded-full px-2.5 py-1 text-[10px] font-black text-white shadow-sm transition group-hover:scale-105" style={{ background: category.color || "var(--brand)" }}>
   394→          {done > 0 ? "Continue" : "Start"}
   395→          <ArrowRight className="h-3 w-3 rtl:rotate-180" />
   396→        </span>
   397→      </motion.button>
   398→    )
   399→  }
   400→
   401→  return (
   402→    <motion.button
