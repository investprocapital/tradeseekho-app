   401→function LessonEditor({ open, id, onOpenChange }: { open: boolean; id: string | null; onOpenChange: (v: boolean) => void }) {
   402→  const t = useT()
   403→  const pick = usePick()
   404→  const { data: cats } = useCategories()
   405→  const { data: detail, isLoading } = useAdminLessonDetail(id)
   406→  const save = useSaveLesson()
   407→  const lang = useStore((s) => s.lang)
   408→  const [locTab, setLocTab] = useState<typeof LANG_ORDER[number]>(lang)
   409→  const [form, setForm] = useState(() => ({
   410→    categoryId: "",
   411→    title: { ...EMPTY_LOC },
   412→    summary: { ...EMPTY_LOC },
   413→    content: { ...EMPTY_LOC },
   414→    imageUrl: "",
   415→    order: 1,
   416→    durationMin: 5,
   417→    isPublished: false,
   418→    isFree: true,
   419→  }))
   420→
   421→  // Reset form when the editing target (id) changes — adjust-during-render (lint-clean)
   422→  const [initedId, setInitedId] = useState<string | null | undefined>(undefined)
   423→  if (id !== initedId) {
   424→    setInitedId(id)
   425→    if (!id) {
   426→      setForm({
   427→        categoryId: cats?.categories[0]?.id ?? "",
   428→        title: { ...EMPTY_LOC },
   429→        summary: { ...EMPTY_LOC },
   430→        content: { ...EMPTY_LOC },
   431→        imageUrl: "",
   432→        order: 1,
   433→        durationMin: 5,
   434→        isPublished: false,
   435→        isFree: true,
   436→      })
   437→    }
   438→    setLocTab(lang)
   439→  }
   440→  // Populate form once server detail arrives (editing mode)
   441→  const [initedDetailId, setInitedDetailId] = useState<string | null>(null)
   442→  if (detail && detail.lesson.id !== initedDetailId) {
   443→    setInitedDetailId(detail.lesson.id)
   444→    setForm({
   445→      categoryId: detail.lesson.categoryId,
   446→      title: { ...detail.lesson.title },
   447→      summary: { ...detail.lesson.summary },
   448→      content: { ...detail.lesson.content },
   449→      imageUrl: detail.lesson.imageUrl ?? "",
   450→      order: detail.lesson.order,
   451→      durationMin: detail.lesson.durationMin,
   452→      isPublished: detail.lesson.isPublished,
   453→      isFree: true,
   454→    })
   455→    setLocTab(lang)
   456→  }

   500→      imageUrl: form.imageUrl || null,
   501→      order: Number(form.order) || 0,
   502→      durationMin: Number(form.durationMin) || 5,
   503→      isPublished: publish ? true : form.isPublished,
   504→      isFree: form.isFree,
   505→    }
   506→    try {
   507→      await save.mutateAsync({ id: id ?? undefined, data: payload })
   508→      toast.success(publish ? t("toast.lessonPublished") : t("toast.lessonSaved"))
   509→      onOpenChange(false)
   510→    } catch {
   511→      toast.error(t("common.error"))

   517→      <DialogContent className="max-h-[90vh] overflow-y-auto p-0 sm:max-w-2xl">

   520→            <BookOpen className="h-5 w-5 text-brand" />
   521→            {id ? t("action.edit") : t("admin.newLesson")}
   522→          </DialogTitle>
   523→          <DialogDescription>{t("admin.lessonEditor")}</DialogDescription>
   524→        </DialogHeader>
   525→
   526→        <div className="max-h-[70vh] overflow-y-auto p-5">
   527→          <div className="space-y-5 p-5">
   528→            {/* Category + meta */}

   549→                </div>
   550→              </div>
   551→            </div>
   552→
   553→            <div className="space-y-1.5">
   554→              <Label>{t("admin.lessonImage")}</Label>
   555→              <div className="flex gap-2">
   556→                <Input value={form.imageUrl} onChange={(e) => setForm((f) => ({ ...f, imageUrl: e.target.value }))} placeholder="URL or upload below" className="h-10" />

   658→function QuizzesTab() {
   659→  const t = useT()
   660→  const pick = usePick()
   661→  const { data: lessonsData } = useAdminLessons()
   662→  const [lessonId, setLessonId] = useState<string | null>(null)
   663→  const { data: quizData, isLoading } = useAdminQuiz(lessonId)
   664→  const save = useSaveQuiz()
   665→  const lang = useStore((s) => s.lang)
   666→  const [locTab, setLocTab] = useState<typeof LANG_ORDER[number]>(lang)
   667→  const [passMark, setPassMark] = useState(3)
   668→  const [questions, setQuestions] = useState(() => [blankQuestion()])
   669→
   670→  // Sync the editor language tab with the global language (adjust-during-render, lint-clean)
   671→  const [seenLang, setSeenLang] = useState(lang)
   672→  if (lang !== seenLang) {
   673→    setSeenLang(lang)
   674→    setLocTab(lang)
   675→  }
   676→
   677→  // Seed questions + passMark from server when the quiz/lesson target changes
   678→  const [initedKey, setInitedKey] = useState<string | null>(null)
   679→  const quizKey = lessonId
   680→    ? `${lessonId}::${quizData?.quiz?.id ?? "none"}::${isLoading ? "loading" : "done"}`
   681→    : null
   682→  if (quizKey && quizKey !== initedKey) {
   683→    setInitedKey(quizKey)
   684→    if (quizData?.quiz) {
   685→      setPassMark(quizData.quiz.passMark)
   686→      setQuestions(
   687→        quizData.quiz.questions.length
   688→          ? quizData.quiz.questions.map((q) => ({
   689→              type: q.type,
   690→              prompt: { ...q.prompt },
   691→              options: q.options.map((o) => ({ ...o })),
   692→              correctIndex: q.correctIndex,
   693→              explanation: { ...q.explanation },
   694→            }))
   695→          : [blankQuestion()],
   696→      )
   697→    } else {
   698→      setPassMark(3)
   699→      setQuestions([blankQuestion()])
   700→    }
   701→  }
   702→
   703→  const updateQ = (i: number, patch: any) =>
   704→    setQuestions((qs) => qs.map((q, idx) => (idx === i ? { ...q, ...patch } : q)))
   705→
   706→  const setQLoc = (i: number, field: "prompt" | "explanation", l: string, val: string) =>
   707→    setQuestions((qs) => qs.map((q, idx) => (idx === i ? { ...q, [field]: { ...q[field], [l]: val } } : q)))
   708→
   709→  const setOptLoc = (i: number, oi: number, l: string, val: string) =>
   710→    setQuestions((qs) =>
   711→      qs.map((q, idx) =>
   712→        idx === i
   713→          ? { ...q, options: q.options.map((o, oidx) => (oidx === oi ? { ...o, [l]: val } : o)) }
   714→          : q,
   715→      ),
   716→    )
   717→
   718→  const addOption = (i: number) =>
   719→    setQuestions((qs) =>
   720→      qs.map((q, idx) => (idx === i && q.options.length < 4 ? { ...q, options: [...q.options, { ...EMPTY_LOC }] } : q)),
   721→    )
   722→  const removeOption = (i: number, oi: number) =>
   723→    setQuestions((qs) =>
   724→      qs.map((q, idx) => {
   725→        if (idx !== i || q.options.length <= 2) return q
   726→        const options = q.options.filter((_, oidx) => oidx !== oi)
   727→        const correctIndex = q.correctIndex >= options.length ? options.length - 1 : q.correctIndex
   728→        return { ...q, options, correctIndex }
   729→      }),
   730→    )
   731→
   732→  const setType = (i: number, type: "MCQ" | "TF") =>
   733→    setQuestions((qs) =>
   734→      qs.map((q, idx) =>
   735→        idx === i
   736→          ? { ...q, type, options: type === "TF" ? q.options.slice(0, 2) : q.options.length < 2 ? [{ ...EMPTY_LOC }, { ...EMPTY_LOC }] : q.options }
   737→          : q,
   738→      ),
   739→    )
   740→
   741→  const saveQuiz = async () => {
   742→    if (!lessonId) return
   743→    try {
   744→      await save.mutateAsync({ lessonId, body: { passMark, questions } })
   745→      toast.success(t("toast.questionSaved"))
   746→    } catch { toast.error(t("common.error")) }
   747→  }
   748→
   749→  return (
   750→    <div className="space-y-4">
   751→      <Card>
   752→        <CardContent className="flex flex-col gap-3 p-4 sm:flex-row sm:items-end">
   753→          <div className="flex-1 space-y-1.5">
   754→            <Label>{t("admin.quizFor")}</Label>
   755→            <Select value={lessonId ?? ""} onValueChange={(v) => setLessonId(v)}>
   756→              <SelectTrigger className="h-10"><SelectValue placeholder="Select lesson" /></SelectTrigger>
   757→              <SelectContent>
   758→                {lessonsData?.lessons.map((l) => (
   759→                  <SelectItem key={l.id} value={l.id}>{pick(l.title)}</SelectItem>
   760→                ))}
   761→              </SelectContent>
   762→            </Select>
   763→          </div>
   764→          <div className="w-32 space-y-1.5">
   765→            <Label>Pass mark</Label>
   766→            <Input type="number" min={1} value={passMark} onChange={(e) => setPassMark(Number(e.target.value) || 1)} className="h-10" />
   767→          </div>
   768→        </CardContent>
   769→      </Card>
   770→
   771→      {!lessonId ? (
   772→        <Card><CardContent className="p-10 text-center text-sm text-muted-foreground">{t("common.empty")}</CardContent></Card>
   773→      ) : isLoading ? (
   774→        <Card><CardContent className="p-10 text-center text-sm text-muted-foreground">{t("common.loading")}</CardContent></Card>
   775→      ) : (
   776→        <>
   777→          {/* Language tabs */}
   778→          <div className="flex flex-wrap gap-1.5">
   779→            {LANG_ORDER.map((code) => (
   780→              <button key={code} onClick={() => setLocTab(code)} className={`rounded-md px-2.5 py-1 text-xs font-bold transition ${locTab === code ? "bg-brand text-brand-foreground" : "bg-muted text-muted-foreground hover:bg-muted/70"}`}>
   781→                {LANGS[code].native}
   782→              </button>
   783→            ))}
   784→          </div>
   785→
   786→          <div className="space-y-3">
   787→            {questions.map((q, i) => (
   788→              <Card key={i} className="overflow-hidden">
   789→                <CardContent className={`p-4 ${locTab === "ur" || locTab === "ar" ? "font-urdu" : ""}`}>
   790→                  <div className="mb-3 flex items-center justify-between gap-2">
   791→                    <div className="flex items-center gap-2">
   792→                      <span className="inline-flex h-7 min-w-7 items-center justify-center rounded-md bg-brand px-1.5 text-xs font-bold text-brand-foreground">{i + 1}</span>
   793→                      <div className="inline-flex rounded-md border border-border p-0.5">
   794→                        <button onClick={() => setType(i, "MCQ")} className={`rounded px-2 py-0.5 text-[11px] font-bold ${q.type === "MCQ" ? "bg-brand text-brand-foreground" : "text-muted-foreground"}`}>MCQ</button>
   795→                        <button onClick={() => setType(i, "TF")} className={`rounded px-2 py-0.5 text-[11px] font-bold ${q.type === "TF" ? "bg-brand text-brand-foreground" : "text-muted-foreground"}`}>T/F</button>
   796→                      </div>
   797→                    </div>
   798→                    <Button size="icon" variant="ghost" className="h-8 w-8 text-destructive hover:text-destructive" onClick={() => setQuestions((qs) => qs.filter((_, idx) => idx !== i))} aria-label={t("action.delete")}>
   799→                      <Trash2 className="h-4 w-4" />
   800→                    </Button>
   801→                  </div>
   802→
   803→                  <div className="space-y-3">
   804→                    <div className="space-y-1">
   805→                      <Label className="text-xs">{t("admin.questionPrompt")} ({LANGS[locTab].native})</Label>
   806→                      <Textarea value={q.prompt[locTab]} onChange={(e) => setQLoc(i, "prompt", locTab, e.target.value)} rows={2} className="text-sm" />
   807→                    </div>

   809→                      <Label className="text-xs">{t("admin.option")} · {t("admin.correct")} ({LANGS[locTab].native})</Label>

   839→                  </div>
   840→                </CardContent>

   845→          <div className="sticky bottom-0 flex items-center justify-between gap-2 border-t border-border bg-background/90 py-3 backdrop-blur">

   851→            </Button>
   852→          </div>
