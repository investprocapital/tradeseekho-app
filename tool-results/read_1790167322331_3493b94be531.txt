  1306→export async function seedEurUsdLessons(db: PrismaClient) {
  1307→  // Get the Beginner category
  1308→  const beginner = await db.category.findUnique({ where: { slug: "beginner" } })
  1309→  if (!beginner) throw new Error("Beginner category not found — run main seed first")
  1310→
  1311→  // Delete old beginner lessons + their quizzes/questions
  1312→  const oldLessons = await db.lesson.findMany({ where: { categoryId: beginner.id }, select: { id: true } })
  1313→  if (oldLessons.length > 0) {
  1314→    await db.lesson.deleteMany({ where: { id: { in: oldLessons.map((l) => l.id) } } })
  1315→  }
  1316→
  1317→  let lessonCount = 0
  1318→  let quizCount = 0
  1319→
  1320→  for (const ls of LESSONS) {
  1321→    const lesson = await db.lesson.create({
  1322→      data: {
  1323→        id: ls.id,
  1324→        categoryId: beginner.id,
  1325→        order: ls.order,
  1326→        titleEn: ls.title.en, titleUr: ls.title.ur, titleHi: ls.title.hi, titleAr: ls.title.ar,
  1327→        summaryEn: ls.summary.en, summaryUr: ls.summary.ur, summaryHi: ls.summary.hi, summaryAr: ls.summary.ar,
  1328→        contentEn: ls.content.en, contentUr: ls.content.ur, contentHi: ls.content.hi, contentAr: ls.content.ar,
  1329→        imageUrl: `/lessons/lesson-${ls.order}.png`,
  1330→        durationMin: ls.durationMin,
  1331→        isPublished: true,
  1332→        isFree: ls.isFree,
  1333→      },
  1334→    })
  1335→
  1336→    // Create quiz with questions
  1337→    const quiz = await db.quiz.create({
  1338→      data: { lessonId: lesson.id, passMark: 3 },
  1339→    })
  1340→
  1341→    for (const q of ls.questions) {
  1342→      await db.question.create({
  1343→        data: {
  1344→          quizId: quiz.id,
  1345→          type: q.type,
  1346→          promptEn: q.prompt.en, promptUr: q.prompt.ur, promptHi: q.prompt.hi, promptAr: q.prompt.ar,
  1347→          option1En: q.options[0].en, option1Ur: q.options[0].ur, option1Hi: q.options[0].hi, option1Ar: q.options[0].ar,
  1348→          option2En: q.options[1].en, option2Ur: q.options[1].ur, option2Hi: q.options[1].hi, option2Ar: q.options[1].ar,

  1350→          option4En: q.options[3].en, option4Ur: q.options[3].ur, option4Hi: q.options[3].hi, option4Ar: q.options[3].ar,
  1351→          correctIndex: q.correctIndex,
  1352→          explanationEn: q.explanation.en, explanationUr: q.explanation.ur, explanationHi: q.explanation.hi, explanationAr: q.explanation.ar,
  1353→        },
  1354→      })
  1355→    }
  1356→    lessonCount++
  1357→    quizCount++
  1358→  }
  1359→
  1360→  return { lessons: lessonCount, quizzes: quizCount, questions: LESSONS.reduce((a, l) => a + l.questions.length, 0) }
