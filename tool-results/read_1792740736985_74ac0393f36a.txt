    22→interface Signal {
    23→  id: string
    24→  symbol: string
    25→  signalType: string
    26→  entry: string

    28→  stopLoss: string

    35→  // different field names. All point to the same base64 data URL.
    36→  chart_image_url?: string

    38→  status: string // active | tp1_hit | tp2_hit | tp3_hit | sl_hit

    48→
    49→async function fetchSignals() {
    50→  const res = await fetch("/api/signals")
    51→  if (!res.ok) throw new Error("failed")
    52→  return res.json() as Promise<{ signals: Signal[] }>
    53→}
    54→
    55→// Resolve the chart image URL. Checks all three field names the API returns
    56→// (screenshot / chart_image_url / chart_snapshot_url) so any client — web or
    57→// Flutter — gets the captured chart image. Returns "" when none exist.
    58→function chartImage(s: Signal): string {
    59→  return s.screenshot || s.chart_image_url || s.chart_snapshot_url || ""
    60→}

   423→                {selected.note && (

   425→                    {selected.note}

   467→                {commentsEnabled && <CommentSection signalId={selected.id} />}

   469→            </motion.div>
   470→          </motion.div>

   477→/**
   478→ * Fallback "chart" shown in the detail modal when a signal has no captured
   479→ * screenshot image (chart_image_url was null). Renders a styled signal-card
   480→ * with the price-level ladder so the client never shows an empty dark box.
   481→ */
   482→function FallbackSignalCard({ s }: { s: Signal }) {
   483→  const isBuy = s.signalType === "BUY"
   484→  const accent = isBuy ? "#00D09C" : "#FF6B6B"
   485→  const nums = [
   486→    { label: "SL", val: parseFloat(s.stopLoss), color: "#FF6B6B", hit: !!s.slHitAt },
   487→    { label: "ENTRY", val: parseFloat(s.entry), color: "#FFFFFF", hit: false },
   488→    { label: "TP1", val: s.tp1 ? parseFloat(s.tp1) : NaN, color: "#00D09C", hit: !!s.tp1HitAt },
   489→    { label: "TP2", val: s.tp2 ? parseFloat(s.tp2) : NaN, color: "#22D3EE", hit: !!s.tp2HitAt },
   490→    { label: "TP3", val: s.tp3 ? parseFloat(s.tp3) : NaN, color: "#3B82F6", hit: !!s.tp3HitAt },
   491→  ].filter((n) => Number.isFinite(n.val))
   492→
   493→  const min = Math.min(...nums.map((n) => n.val))
   494→  const max = Math.max(...nums.map((n) => n.val))
   495→  const range = Math.max(max - min, 1e-6)
   496→  const pad = range * 0.15
   497→  const lo = min - pad
   498→  const hi = max + pad
   499→  const pct = (v: number) => `${((hi - v) / (hi - lo)) * 100}%`
   500→
   501→  return (
   502→    <div
   503→      className="relative w-full overflow-hidden"
   504→      style={{ background: "linear-gradient(180deg,#0B1B2E 0%,#0A1929 100%)" }}
   505→    >
   506→      <div style={{ height: 4, background: accent }} />
   507→      <div className="p-4">
   508→        {/* Header: pair + BUY/SELL pill */}
   509→        <div className="flex items-center justify-between">
   510→          <span className="text-2xl font-extrabold text-white">{pairLabel(s.symbol)}</span>
   511→          <span
   512→            className="rounded-full px-3 py-1 text-xs font-bold"
   513→            style={{ background: accent, color: "#0A1929" }}
   514→          >
   515→            {s.signalType}
   516→          </span>
   517→        </div>
   518→        {/* Ladder */}
   519→        <div className="relative mt-4 mb-2" style={{ height: 200 }}>
   520→          <div className="absolute left-3 top-0 bottom-0 w-0.5 bg-[#1E3A5F]" />
   521→          {nums.map((n) => (
   522→            <div
   523→              key={n.label}
   524→              className="absolute flex items-center"
   525→              style={{ top: pct(n.val), left: 0, right: 0 }}
   526→            >
   527→              <div className="h-0.5 flex-1" style={{ background: n.color }} />
   528→              <div
   529→                className="ms-auto me-3 rounded px-2 py-0.5 text-[11px] font-bold"
   530→                style={{ background: n.color, color: "#0A1929" }}
   531→              >
   532→                {n.label}: {n.val} {n.hit ? "✅" : ""}
   533→              </div>
   534→            </div>
   535→          ))}
   536→        </div>
   537→        <div className="mt-1 text-center text-[10px] text-white/40">
   538→          TradeSeekho PK · LEARN TRADE GROW
   539→        </div>
   540→      </div>
   541→    </div>
   542→  )
   543→}
